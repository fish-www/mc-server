# ============================================
# Minecraft Server JVM 参数配置
# GraalVM 25
# ============================================

# === 内存配置 (必须 Xms = Xmx 以支持大页面与 PreTouch) ===
-Xmx54G
-Xms54G
-XX:+AlwaysPreTouch

# timeout
-Dminecraft.api.auth.timeout=60
-Dsun.rmi.transport.connectionTimeout=120000
-Dnetty.timeOut=120

# === 硬件与架构优化 (NUMA & Large Pages) ===
-XX:+UseNUMA
-XX:+UseLargePages
-XX:LargePageSizeInBytes=2m
-XX:+UnlockExperimentalVMOptions
-XX:+UnlockDiagnosticVMOptions
-XX:+UseCriticalJavaThreadPriority
-XX:+AlwaysActAsServerClassMachine

# === 分代 ZGC 特化调优 ===
-XX:+UseZGC
-XX:ZAllocationSpikeTolerance=5
-XX:+PerfDisableSharedMem
-XX:+UseStringDeduplication

# === JIT 编译器基础优化 ===
-XX:-DontCompileHugeMethods
-XX:MaxNodeLimit=240000
-XX:NodeLimitFudgeFactor=8000
-XX:NmethodSweepActivity=1
-XX:+EagerJVMCI

# === 代码缓存配置 (满足大模组服/高并发) ===
-XX:ReservedCodeCacheSize=1024M
-XX:NonNMethodCodeHeapSize=32M
-XX:ProfiledCodeHeapSize=496M
-XX:NonProfiledCodeHeapSize=496M

# === 自动向量化与高级指令集 ===
-XX:+AlignVector
-XX:+OptimizeFill
-XX:+EnableVectorSupport
-XX:+UseCharacterCompareIntrinsics
-XX:+UseVectorStubs
-Djava.security.egd=file:/dev/urandom

# === GraalVM EE 编译器配置 ===
-Djdk.graal.CompilerConfiguration=enterprise
-Djdk.graal.UsePriorityInlining=true
-Djdk.graal.TuneInlinerExploration=0

# === GraalVM 循环与内联深度优化 ===
-Djdk.graal.LoopInversion=true
-Djdk.graal.LoopRotation=true
-Djdk.graal.DetectInvertedLoopsAsCounted=true
-Djdk.graal.EnterprisePartialUnroll=true

# === GraalVM SIMD 矢量化优化 ===
-Djdk.graal.Vectorization=true
-Djdk.graal.VectorizeSIMD=true
-Djdk.graal.VectorizeHashes=true
